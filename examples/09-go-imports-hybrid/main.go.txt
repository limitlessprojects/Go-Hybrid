// 09-go-imports-hybrid — THE ADOPTION STORY. This side is plain Go: it imports
// a Hybrid package and needs NO Hybrid toolchain (generated Go source + a
// prebuilt Rust artifact ship with the package — verified feasible by spike 2).
// TARGET SYNTAX. File: main.go (a Go file, not .hy!).

package main

import (
	"fmt"

	"example.com/app/hybrid" // the Hybrid-compiled package
)

func main() {
	email, err := hybrid.ValidateEmail("ada@lovelace.dev")
	fmt.Println(email, err)

	// calls THROUGH to the Rust regex crate the Hybrid package pinned
	ok := hybrid.SimplePattern("ab*c") 
	fmt.Println("match:", ok)
}
