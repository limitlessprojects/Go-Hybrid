package capbench

// ---- Hybrid prelude: capacity clipping (auto-generated) ----
func hy_clip[s ~[]e, e any](view s) s {
	return view[:len(view):len(view)]
}

type store struct {
	data [1024]int
}

func View(s []int, lo int, hi int) []int {
	return hy_clip(s[lo:hi])
}

func Prefix(s []int, hi int) []int {
	return hy_clip(s[:hi])
}

func Suffix(s []int, lo int) []int {
	return hy_clip(s[lo:])
}

func Whole(s []int) []int {
	return hy_clip(s[:])
}

func ArrayView(s *store, lo int, hi int) []int {
	return hy_clip(s.data[lo:hi])
}

func PointerView(s *[1024]int, lo int, hi int) []int {
	return hy_clip(s[lo:hi])
}

func Nested(s []int, lo int, hi int) []int {
	return hy_clip(hy_clip(s[:hi])[lo:])
}
